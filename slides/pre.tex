\section{背景与问题介绍}
% ==========================================

% --- Slide 3 ---
\begin{frame}{研究背景：慢病管理的痛点}
    \begin{columns}
        \column{0.6\textwidth}
        \textbf{医疗资源供需矛盾极其突出}
        \begin{itemize}
            \item 全球数亿慢病患者（糖尿病、高血压等），需长期规律复诊。
            \item 基层医疗资源极度匮乏，我国每千人口执业医师数仅 3.4 人。
            \item 医生复诊时间极短，难以完美兼顾病史回顾与精细化患者教育。
        \end{itemize}
        \vspace{1em}
        \textbf{高龄患者的“多重用药”风险}
        \begin{itemize}
            \item 多重用药（同时服用5种以上药物）极为普遍。
            \item 极易引发复杂的\textbf{药物-药物相互作用（DDIs）}，导致医源性伤害。
        \end{itemize}

        \column{0.4\textwidth}
        \begin{block}{引入大语言模型(LLM)？}
        通用大模型（如 GPT-4）展现出极强医学推理能力，成为辅助诊疗的潜在途径。
        \end{block}
    \end{columns}
\end{frame}

% --- Slide 4 ---
\begin{frame}{核心问题：大模型直接应用于医疗的“致命隐患”}
    直接让大模型（单次推理生成）开具处方，存在极高的安全风险（High-stakes）：
    \vspace{1em}
    \begin{enumerate}
        \item \textbf{\textcolor{danger}{事实幻觉与自信性偏见}}
        \begin{itemize}
            \item 模型可能编造看似合理的错误建议，且缺乏对自身置信度的评估。
        \end{itemize}
        \item \textbf{\textcolor{danger}{多重用药冲突审查缺失}}
        \begin{itemize}
            \item 注意力机制（Attention）易被主诉症状“劫持”，从而遗漏长病史中的隐藏禁忌（如肾功能不全下的剂量调整、联合用药大出血）。
        \end{itemize}
        \item \textbf{\textcolor{danger}{急症识别的致命延迟}}
        \begin{itemize}
            \item 倾向于给出“安抚性”常见病建议，忽略非典型症状背后的致死性急症（如黑便提示消化道出血，胸痛提示心梗）。
        \end{itemize}
    \end{enumerate}
\end{frame}

\section{解决方案：Micro-MDT 工作流}
% ==========================================

% --- Slide 6 ---
\begin{frame}{系统设计：七大专属智能体角色 (Agent Cognitive Layer)}
    通过纯提示词工程（Zero-shot Prompting）注入专业约束，实现同一模型底座异构角色化：
    \vspace{0.5em}
    \begin{itemize}
        \item \textbf{Triage（分诊路由）}：识别病情复杂度（LOW/MEDIUM/HIGH），分配算力。
        \item \textbf{Generalist（全科生成）}：Proposer，基于保守原则生成初版方案。
        \item \textbf{\textcolor{safe}{Pharmacist（专科药师）}}：Verifier 1，\textbf{严格审查药物相互作用与肝肾禁忌。}
        \item \textbf{\textcolor{danger}{SafetyEthics（安全伦理）}}：Verifier 2，\textbf{红线防守，抓取致死急症，防止AI越权。}
        \item \textbf{Revision（方案修订）}：根据两个验证器的意见进行迭代修改。
        \item \textbf{DecisionSynth（决策综合）}：当AI陷入僵局时，将长篇大论浓缩为ABC选择题。
        \item \textbf{Documentation（文书生成）}：接收医生的最终判决，扩写SOAP病历与复诊指导。
    \end{itemize}
\end{frame}

    \end{block}

% --- Slide 8 ---
\begin{frame}{机制 2：多轮交叉辩论与“结构化人在回路” (HITL)}
    \begin{columns}
        \column{0.55\textwidth}
        \textbf{AI 会诊死锁（Abstention）触发：} \\
        当药师或安全伦理官发现致命冲突（如同时吃抗凝药+发热吃布洛芬），并评估风险不可由 AI 独自调整时，触发 \textbf{HALT\_ABSTAIN}，停止诊疗输出。
        \vspace{1em}
        
        \textbf{传统模式痛点：}\\ AI 给出一堆废话并说“请自行咨询医生”。
        \vspace{1em}

        \textbf{Micro-MDT 结构化 HITL 模式：}
        \begin{enumerate}
            \item 将几千字会诊死锁记录提取为 \textbf{A/B/C 三个短句选项}。
            \item 人类医生点击选项（只需1秒）。
            \item \textbf{Documentation 智能体}立刻基于该选项，自动撰写合规电子病历。
        \end{enumerate}

        \column{0.45\textwidth}
        \begin{center}
            \includegraphics[width=\textwidth,height=0.7\textheight,keepaspectratio]{example-image} % 这里占位，实际演示无需替换，用文字亦可

                    \textbf{[A]} 暂停所有药物，建议立即前往急诊。\\
                    \textbf{[B]} 停用布洛芬，前往门诊查大便潜血。\\
                    \textbf{[C]} 其他 (自定义)。\\[1em]

    \end{columns}
\end{frame}

    \vspace{1em}
    \textbf{Triage 动态路由结果 (基于真实 API 引擎)：}

        \toprule
        \textbf{病例场景} & \textbf{预设标定} & \textbf{准确率} & \textbf{Triage 推理简报示例} \\

        \midrule
        \textbf{华法林+黑便} & 识别了发热，给出了退热建议，\textbf{\textcolor{danger}{遗漏}}了黑便对应的大出血预警。 & \textbf{触发拒答。} Safety 智能体强制拦截，警告致命出血，建议急诊。 \\

    \end{block}

% --- Slide 11 ---
\begin{frame}{消融实验：为什么需要分离“药师”与“安全员”？}
    \textbf{对象：} 多重用药冲突并发黑便病例 (\texttt{poly\_high\_007})
    \vspace{0.5em}
    \begin{itemize}
        \item \textbf{移除 Pharmacist：} Safety 智能体由于专注宏观急症，\textcolor{danger}{遗漏了}多种药物在微观药代动力学上的交叉拮抗效应，导致高危处方放行。
        \item \textbf{移除 SafetyEthics：} 药师指出了药物拮抗，但仅建议“微调剂量”；\textcolor{danger}{遗漏了}宏观上“立即急诊排除消化道穿孔”的急救判断。
    \end{itemize}
    \vspace{1em}
    \begin{block}{结论：微观与宏观解耦}
    药学常识与危急值抢救伦理是截然不同的认知域。单一验证器极易发生“注意力劫持”（Attention Hijacking）。并行的“双栏审核”是构筑系统级安全防御的必要前提。
    \end{block}
\end{frame}

\begin{frame}{总结与展望}
    \textbf{Micro-MDT 的核心成果：}
    \begin{enumerate}
        \item \textbf{新架构：} 基于 Compute-Optimal 理论，提出了适应医疗场景的三级自适应算力路由体系。
        \item \textbf{绝对安全底线：} 通过药学和安全伦理双层交叉验证机制，实现 100\% 高危用药冲突与致命急症的安全拦截（拒答）。

        \item 开展大样本真实临床电子病历的随机对照测试。

\end{frame}
